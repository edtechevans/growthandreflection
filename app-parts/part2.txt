  function navigate(name){
    views.forEach(v => v.classList.toggle('is-active', v.id === `view-${name}`));
    bottomButtons.forEach(b => b.classList.toggle('is-active', b.dataset.nav === name));
    window.scrollTo({top:0, behavior:'smooth'});
    if(name === 'reflect' && !activeMode && resultScreen.hidden) resetReflect();
  }

  navButtons.forEach(b => b.addEventListener('click', () => navigate(b.dataset.nav)));
  $$('[data-start]').forEach(b => b.addEventListener('click', () => { navigate('reflect'); openMode(b.dataset.start); }));
  $$('[data-mode]').forEach(b => b.addEventListener('click', () => openMode(b.dataset.mode)));
  $('#changeMode').addEventListener('click', resetReflect);
  $('#newReflection').addEventListener('click', resetReflect);

  function openMode(mode){
    activeMode = mode;
    captureChooser.hidden = true;
    captureWorkspace.hidden = false;
    analysisScreen.hidden = true;
    resultScreen.hidden = true;
    $$('.capture-panel').forEach(p => p.hidden = p.dataset.panel !== mode);
    focusQuestion.value = '';
    if(mode === 'audio') setTimeout(() => $('#recordAudio').focus(), 50);
    window.scrollTo({top:0, behavior:'smooth'});
  }

  function resetReflect(){
    activeMode = null;
    captureChooser.hidden = false;
    captureWorkspace.hidden = true;
    analysisScreen.hidden = true;
    resultScreen.hidden = true;
    $$('.capture-panel').forEach(p => p.hidden = true);
    focusQuestion.value = '';
    $$('.focus-chip,.prompt-chip').forEach(c => c.classList.remove('is-selected'));
  }

  $$('.focus-chip').forEach(chip => chip.addEventListener('click', () => {
    chip.classList.toggle('is-selected');

  }));

  }));
