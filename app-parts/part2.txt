  function navigate(name){
    views.forEach(v=>v.classList.toggle('is-active',v.id===`view-${name}`));
    desktopNav.forEach(b=>b.classList.toggle('is-active',b.dataset.nav===name));
    mobileNav.forEach(b=>b.classList.toggle('is-active',b.dataset.nav===name));
    window.scrollTo({top:0,behavior:'smooth'});
    if(name==='reflect'&&!activeMode&&resultScreen.hidden) resetReflect();
  }
  $$('[data-nav]').forEach(b=>b.addEventListener('click',()=>navigate(b.dataset.nav)));
  $$('[data-start]').forEach(b=>b.addEventListener('click',()=>{navigate('reflect');openMode(b.dataset.start)}));
  $$('[data-mode]').forEach(b=>b.addEventListener('click',()=>openMode(b.dataset.mode)));

  function openMode(mode){
    $('#view-reflect').classList.remove('showing-result');
    activeMode=mode;captureChooser.hidden=true;captureWorkspace.hidden=false;analysisScreen.hidden=true;resultScreen.hidden=true;
    $$('.capture-panel').forEach(p=>p.hidden=p.dataset.panel!==mode);
    const labels={video:'Classroom video',audio:'Voice reflection',file:'Unit / lesson / assessment',write:'Written reflection'};
    $('#evidenceModeLabel').textContent=labels[mode];focusQuestion.value='';
    if(mode==='audio')setTimeout(()=>$('#recordAudio').focus(),60);
    window.scrollTo({top:0,behavior:'smooth'});
  }
  function resetReflect(){
